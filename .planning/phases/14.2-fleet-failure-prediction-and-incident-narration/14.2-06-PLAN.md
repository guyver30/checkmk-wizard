  { id: string, source: "failure"|"trend"|"sustained", host: string, service: string, metric: string, unit: string,
    tier: NeedTier, computed_tier: NeedTier,
    days_to_warn: number|null, days_to_crit: number|null, warn_date: string|null, crit_date: string|null,
    confidence: "low"|"medium"|"high"|null, history_days: number|null,
    value: number|null, warn: number|null, crit: number|null,

    since: string /* ISO UTC */, narration: string,

                     computed_tier_at_set: NeedTier, note: string, by: string },
    generated_at: string }
  NeedTier = "immediate" | "urgent" | "standard"

ForecastPayload (lan/forecasts/{host}, retained, republished each 15-min cycle):
  { host: string, generated_at: string, fits: FitPayload[] }
  FitPayload = { service, metric, unit, status: "trending"|"stable"|"no_clear_trend",
    slope_per_day: number|null, value_at_end: number|null, fit_start_ts: number|null, fit_end_ts: number|null,
    r2: number|null, history_days: number, confidence: "low"|"medium"|"high"|null, last_value: number|null,
    warn: number|null, crit: number|null, warn_ts: number|null, crit_ts: number|null,
    warn_date: string|null, crit_date: string|null, days_to_warn: number|null, days_to_crit: number|null }
  (timestamps are epoch SECONDS)

NarrationPayload (lan/incidents/{incident_id}/narration, retained, tombstone on close):
  { id: string, headline: string, sentences: string[], tier: NeedTier|null, generated_at: string }
